-- =========================================
-- STEP 1: Create Tables
-- =========================================

CREATE TABLE users (
    user_id INTEGER PRIMARY KEY,
    name TEXT,
    signup_date DATE
);

CREATE TABLE orders (
    order_id INTEGER PRIMARY KEY,
    user_id INTEGER,
    amount INTEGER
);

CREATE TABLE activity (
    user_id INTEGER,
    login_count INTEGER
);

-- =========================================
-- Query 1
-- Retrieve each user's name and order amount
-- =========================================

SELECT u.name, o.amount
FROM users u
INNER JOIN orders o
ON u.user_id = o.user_id;


-- =========================================
-- Query 2
-- User name and total order amount
-- =========================================

SELECT u.name, SUM(o.amount) AS total_order_amount
FROM users u
INNER JOIN orders o
ON u.user_id = o.user_id
GROUP BY u.name;


-- =========================================
-- Query 3
-- All users with login counts (including no activity)
-- =========================================

SELECT u.name, a.login_count
FROM users u
LEFT JOIN activity a
ON u.user_id = a.user_id;


-- =========================================
-- Query 4
-- Users who never placed an order
-- =========================================

SELECT u.user_id, u.name
FROM users u
LEFT JOIN orders o
ON u.user_id = o.user_id
WHERE o.order_id IS NULL;


-- =========================================
-- Query 5
-- Combine all three tables
-- =========================================

SELECT 
    u.user_id,
    u.name,
    SUM(o.amount) AS total_order_amount,
    a.login_count
FROM users u
LEFT JOIN orders o
ON u.user_id = o.user_id
LEFT JOIN activity a
ON u.user_id = a.user_id
GROUP BY u.user_id, u.name, a.login_count;


-- =========================================
-- Query 6
-- Users who placed orders greater than 400
-- =========================================

SELECT u.name, o.amount
FROM users u
INNER JOIN orders o
ON u.user_id = o.user_id
WHERE o.amount > 400;


-- =========================================
-- Query 7
-- Top 3 users by total order amount
-- =========================================

SELECT 
    u.name,
    SUM(o.amount) AS total_order_amount
FROM users u
INNER JOIN orders o
ON u.user_id = o.user_id
GROUP BY u.name
ORDER BY total_order_amount DESC
LIMIT 3;


-- =========================================
-- Query 8
-- Sort users by total order amount and login count
-- =========================================

SELECT 
    u.user_id,
    u.name,
    SUM(o.amount) AS total_order_amount,
    a.login_count
FROM users u
LEFT JOIN orders o
ON u.user_id = o.user_id
LEFT JOIN activity a
ON u.user_id = a.user_id
GROUP BY u.user_id, u.name, a.login_count
ORDER BY total_order_amount DESC, a.login_count DESC;


-- =========================================
-- Query 9
-- Average order amount per user
-- =========================================

SELECT 
    u.name,
    AVG(o.amount) AS avg_order_amount
FROM users u
INNER JOIN orders o
ON u.user_id = o.user_id
GROUP BY u.name;


-- =========================================
-- Query 10
-- Combined AI feature query
-- =========================================

SELECT 
    u.user_id,
    u.name,
    SUM(o.amount) AS total_order_amount,
    a.login_count
FROM users u
LEFT JOIN orders o
ON u.user_id = o.user_id
LEFT JOIN activity a
ON u.user_id = a.user_id
GROUP BY u.user_id, u.name, a.login_count
ORDER BY total_order_amount DESC
LIMIT 3;
