-- =========================================
-- Step 1: Create the students table
-- =========================================

CREATE TABLE students (
    id INTEGER PRIMARY KEY,
    name TEXT,
    score INTEGER,
    passed BOOLEAN,
    class INTEGER
);

-- =========================================
-- Query 1: Retrieve all columns
-- =========================================

SELECT *
FROM students;


-- =========================================
-- Query 2: Retrieve only name and score
-- =========================================

SELECT name, score
FROM students;


-- =========================================
-- Query 3: Students who scored above 80
-- =========================================

SELECT *
FROM students
WHERE score > 80;


-- =========================================
-- Query 4: Students who scored above 80 AND passed
-- =========================================

SELECT *
FROM students
WHERE score > 80 AND passed = TRUE;


-- =========================================
-- Query 5: Students who scored above 85 OR belong to class 9
-- =========================================

SELECT *
FROM students
WHERE score > 85 OR class = 9;


-- =========================================
-- Query 6: Students who did NOT pass
-- =========================================

SELECT *
FROM students
WHERE NOT passed;


-- =========================================
-- Query 7: List of distinct classes
-- =========================================

SELECT DISTINCT class
FROM students;


-- =========================================
-- Query 8: Top 5 highest scoring students
-- =========================================

SELECT *
FROM students
ORDER BY score DESC
LIMIT 5;


-- =========================================
-- Query 9: Sort by class ASC and score DESC
-- =========================================

SELECT *
FROM students
ORDER BY class ASC, score DESC;


-- =========================================
-- Query 10: Rename score using alias
-- =========================================

SELECT name, score AS final_score
FROM students;


-- =========================================
-- Query 11: Combined Query
-- Select name and score
-- Score > 75
-- Sort by score DESC
-- Top 3 results
-- =========================================

SELECT name, score
FROM students
WHERE score > 75
ORDER BY score DESC
LIMIT 3;
